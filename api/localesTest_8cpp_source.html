26void LocalesTest::testInitWithTranslations() {

36void LocalesTest::testLocaleSwitching() {
37 // the force flag reloads the translations of the same locale.
38 QVERIFY(Locales::setLocale(QLocale(QLocale::English, QLocale::AnyCountry), true));
39 QCOMPARE(Locales::currentLocate(), QLocale(QLocale::English, QLocale::AnyCountry));

41 // a locale without an available translation file must return false.
