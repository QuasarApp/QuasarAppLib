44void GlobalUtilsTest::testHashOperations() {
45 // the subtract operation removes the right hash keys from the left hash.
46 {
47 QHash<QString, int> left{{"a", 1}, {"b", 2}, {"c", 3}};
48 QHash<QString, int> right{{"b", 9}};
49 subtract(left, right);
50 QCOMPARE(left.size(), 2);
51 QVERIFY(!left.contains("b"));
52 QVERIFY(left.contains("a"));
53 QVERIFY(left.contains("c"));
54 }
55
56 // the subtract operation clears the left hash when left and right
57 // share the same data. The clear operation detaches the shared data
58 // (copy on write), so the other copy keeps its values.
59 {
60 QHash<QString, int> left{{"a", 1}, {"b", 2}};
61 QHash<QString, int> right = left;
62 QVERIFY(left.isSharedWith(right));
63 subtract(left, right);
64 QVERIFY(left.isEmpty());
65 QCOMPARE(right.size(), 2);
66 }
67
68 // the unite operation inserts the right hash values into the left hash.
69 {
70 QHash<QString, int> left{{"a", 1}};
71 QHash<QString, int> right{{"a", 3}, {"b", 2}};
72 unite(left, right);
73 QCOMPARE(left.size(), 2);
74 QCOMPARE(left.value("a"), 3);
75 QCOMPARE(left.value("b"), 2);
76 }
77
78 // the unite operation is skipped when left and right share the same data.
79 {
80 QHash<QString, int> left{{"a", 1}};
81 QHash<QString, int> right = left;
82 QVERIFY(left.isSharedWith(right));
83 unite(left, right);
84 QCOMPARE(left.size(), 1);
85 QCOMPARE(right.size(), 1);
86 }
87
88 // the match operation keeps only the keys that exists in the right hash.
89 {
90 QHash<QString, int> left{{"a", 1}, {"b", 2}, {"c", 3}};
91 QHash<QString, int> right{{"b", 99}, {"c", 99}};
92 match(left, right);
93 QCOMPARE(left.size(), 2);
94 QVERIFY(left.contains("b"));
95 QVERIFY(left.contains("c"));
96 QVERIFY(!left.contains("a"));
97 // the values of the left hash are not changed by the match operation.
98 QCOMPARE(left.value("b"), 2);
99 QCOMPARE(left.value("c"), 3);
100 }
101}

112void GlobalUtilsTest::testTypeHashes() {
113 // all type based hashes are derived from the std::type_info hash code
114 // of the type. The 32 bit version is the truncated (32 bit) value and
115 // the 16/8 bit versions are the remainders of the full hash code.
116 const std::size_t typeHash = typeid(int).hash_code();
117 QCOMPARE(static_type_hash_32<int>(), static_cast<uint32_t>(typeHash));
118 QCOMPARE(static_type_hash_16<int>(), static_cast<uint16_t>(typeHash % 0xFFFF));
119 QCOMPARE(static_type_hash_8<int>(), static_cast<uint8_t>(typeHash % 0xFF));
120
121 // the object based hash must be equal to the type based hash.
122 int object = 1;
123 QCOMPARE(static_type_hash_32(object), static_type_hash_32<int>());
124 QCOMPARE(static_type_hash_16(object), static_type_hash_16<int>());
125 QCOMPARE(static_type_hash_8(object), static_type_hash_8<int>());
126
127 // different types must have different hashes.
128 QVERIFY(static_type_hash_32<int>() != static_type_hash_32<QString>());
129}
