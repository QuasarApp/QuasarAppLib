121void ConfigTest::testPriorityOrder() {
122 // 1. The command-line value has the highest priority.
123 {
124 const char* argv[] = {"test", "-testValue", "cliValue"};
125 Config config(3, argv, configOptions());
126 QCOMPARE(config.getValue("testValue"), QVariant("cliValue"));
127 }
128
129 // 2. A value saved in the config file doesn't override the
130 // command-line value, but it is used if the command line
131 // doesn't provide the value.
132 {
133 const char* argv[] = {"test"};
134 Config config(1, argv);
135 config.setValue("testValue", QString("fileValue"));
136 config.sync();
137 }
138 {
139 const char* argv[] = {"test", "-testValue", "cliValue2"};
140 Config config(3, argv, configOptions());
141 QCOMPARE(config.getValue("testValue"), QVariant("cliValue2"));
142 }
143 {
144 const char* argv[] = {"test"};
145 Config config(1, argv);
146 QCOMPARE(config.getValue("testValue"), QVariant("fileValue"));
147
148 // 3. The default value (the def argument) has the lowest
149 // priority: it is used only if the key is not set on the
150 // command line and in the config file.
151 QCOMPARE(config.getValue("testValue", QVariant("ignoredDefault")),
152 QVariant("fileValue"));
153 QCOMPARE(config.getValue("absentKey", QVariant("fallback")),
154 QVariant("fallback"));
155 }

216void ConfigTest::testSettingsInheritance() {
217 const char* argv[] = {"test"};
218 TestConfig config(1, argv);
219
220 // the Config object is a Settings object and an ISettings object.
221 Settings* base = &config;
222 QVERIFY(base == static_cast<Settings*>(&config));
223 ISettings* iface = &config;
224 QVERIFY(iface == static_cast<ISettings*>(&config));
225
226 // the Settings base class backend is available.
227 QVERIFY(config.settings() != nullptr);
228
229 // the ISettings signals work through the Config object. The
230 // signals are resolved on the runtime metaobject (see the
231 // settingsTest for the explanation).
232 const QMetaObject* mo = config.metaObject();
233 const int valueIndex = mo->indexOfSignal("valueChanged(QString,QVariant)");
234 const int strIndex = mo->indexOfSignal("valueStrChanged(QString,QString)");
235 QVERIFY(valueIndex >= 0);
236 QVERIFY(strIndex >= 0);
237
238 QSignalSpy valueSpy(&config, mo->method(valueIndex));
239 QSignalSpy strSpy(&config, mo->method(strIndex));
240
241 config.setValue("signalKey", "signalValue");
242 QCOMPARE(valueSpy.count(), 1);
243 QCOMPARE(valueSpy.at(0).at(0).toString(), QString("signalKey"));
244 QCOMPARE(valueSpy.at(0).at(1).toString(), QString("signalValue"));
245 QCOMPARE(strSpy.count(), 1);
246 QCOMPARE(strSpy.at(0).at(0).toString(), QString("signalKey"));
247 QCOMPARE(strSpy.at(0).at(1).toString(), QString("signalValue"));
248
249 // the Settings base class methods work through the Config object.
250 config.setBoolOptions({"boolOpt"});
251 QVERIFY(config.boolOptions().contains("boolOpt"));
