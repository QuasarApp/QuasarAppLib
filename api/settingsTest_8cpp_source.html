116void SettingsTest::testSignals() {
117 auto instance = ISettings::instance();
118
119 // The signals are resolved on the runtime metaobject of the service
120 // instance and connected through the QMetaMethod based QSignalSpy
121 // constructor. A pointer-to-member connect of a signal that is defined
122 // in a shared library does not work from an application on Windows:
123 // the address of the signal function imported by the application
124 // differs from the address stored in the metaobject signal table of
125 // the library, so the connect fails with "signal not found".
126 const QMetaObject *mo = instance->metaObject();
127 const int valueIndex = mo->indexOfSignal("valueChanged(QString,QVariant)");
128 const int strIndex = mo->indexOfSignal("valueStrChanged(QString,QString)");
129 QVERIFY(valueIndex >= 0);
130 QVERIFY(strIndex >= 0);
131
132 QSignalSpy valueSpy(instance, mo->method(valueIndex));
133 QSignalSpy strSpy(instance, mo->method(strIndex));
134
135 instance->setValue("signalKey", "signalValue");
136
137 QCOMPARE(valueSpy.count(), 1);
138 QCOMPARE(valueSpy.at(0).at(0).toString(), QString("signalKey"));
139 QCOMPARE(valueSpy.at(0).at(1).toString(), QString("signalValue"));
140
141 QCOMPARE(strSpy.count(), 1);
142 QCOMPARE(strSpy.at(0).at(0).toString(), QString("signalKey"));
143 QCOMPARE(strSpy.at(0).at(1).toString(), QString("signalValue"));
144}

146void SettingsTest::testSaveModes() {
147 auto instance = ISettings::instance();
148 QCOMPARE(instance->getMode(), SettingsSaveMode::Auto);
149
150 instance->setMode(SettingsSaveMode::Manual);
151 QCOMPARE(instance->getMode(), SettingsSaveMode::Manual);
152
153 instance->setStrValue("manualKey", "manualValue");
154
155 // in the manual mode the value must not be saved on disk until the sync call.
156 QSettings backend(QSettings::IniFormat, QSettings::Scope::UserScope,
157 QCoreApplication::organizationName(),
158 QCoreApplication::applicationName());
159 QVERIFY(backend.value("manualKey").isNull());
160
161 instance->sync();
162
163 // reload the backend from disk before the check, because the QSettings
164 // object was created before the sync call.
165 backend.sync();
166 QCOMPARE(backend.value("manualKey").toString(), QString("manualValue"));
167
168 instance->setMode(SettingsSaveMode::Auto);
169}

186void SettingsTest::testBoolOptions() {
187 QVERIFY(_testSettings != nullptr);
188
189 QVERIFY(_testSettings->boolOptions().isEmpty());
190 _testSettings->setBoolOptions({"boolKey"});
191 QVERIFY(_testSettings->boolOptions().contains("boolKey"));
192
193 // the service instance writes the value into the QSettings backend (auto mode).
194 ISettings::instance()->setValue("boolKey", 1);
195 QVERIFY(ISettings::instance()->getValue("boolKey").toBool());
196
197 // a new settings object with the same storage must read the value as bool,
198 // because the key is registered in the boolOptions map.
199 TestSettings localSettings;
200 localSettings.setBoolOptions({"boolKey"});
201 QVERIFY(localSettings.getValue("boolKey").toBool());
202
203 // now the stored value must be read as false.
204 ISettings::instance()->setValue("boolKey", 0);
205
206 TestSettings localSettings2;
207 localSettings2.setBoolOptions({"boolKey"});
208 QVERIFY(!localSettings2.getValue("boolKey").toBool());
209}
