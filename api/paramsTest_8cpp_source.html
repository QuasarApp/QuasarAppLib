42void ParamsTest::testVerboseParsing() {
43 QVERIFY(Params::parseParams(QStringList{"-verbose", "3"}));
44 QCOMPARE(Params::getArg("verbose"), QString("3"));
45 QCOMPARE(Params::getVerboseLvl(), VerboseLvl::Debug);
46 QVERIFY(Params::isDebug());
47 QCOMPARE(Params::size(), 1);
48 QCOMPARE(Params::getUserParamsMap().value("verbose"), QString("3"));
49
50 QVERIFY(Params::parseParams(QStringList{"-verbose", "1"}));
51 QCOMPARE(Params::getVerboseLvl(), VerboseLvl::Warning);
52 QVERIFY(!Params::isDebug());
53}

166void ParamsTest::testParseParamsPositionalArguments() {
167 // an argument without the leading dash is stored as a positional
168 // argument with the empty value.
169 QVERIFY(Params::parseParams(QStringList{"hello"}));
170 QVERIFY(Params::isEndable("hello"));

172 QCOMPARE(Params::size(), 1);

222void ParamsTest::testParseParamsValueLookingLikeOption() {
223 // the QCommandLineParser consumes the next argument as the option value
224 // as it is, a value can look like an option. The "-verbose" value is a
225 // registered option name, so it is mapped to the double dash form by
226 // the parser path.
227 QVERIFY(Params::parseParams(QStringList{"-fileLog", "-verbose"}));
228 QCOMPARE(Params::getArg("fileLog"), QString("--verbose"));
229
230 // the same for the equals form of the option.
231 QVERIFY(Params::parseParams(QStringList{"-fileLog=-verbose"}));
232 QCOMPARE(Params::getArg("fileLog"), QString("-verbose"));
233
234 // a negative number can be used as an option value.
235 QVERIFY(Params::parseParams(QStringList{"-verbose", "-1"}));
236 QCOMPARE(Params::getArg("verbose"), QString("-1"));
237}
