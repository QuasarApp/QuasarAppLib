14Config::Config(int argc, const char* argv[], const OptionsDataList& options)
15{
16 // All the command-line parsing logic is already implemented in the
17 // Params class, so the Config does not use the QCommandLineParser
18 // directly.
19 if (!Params::parseParams(argc, argv, options)) {
20 qWarning() << "The Config command-line arguments parse error. "
21 "The command-line values will be ignored.";
22 return;
23 }
24
25 // The command-line values have the highest priority, so they are
26 // stored in the snapshot map and checked first in the
27 // getValueImplementation method.
28 _paramsMap = Params::getUserParamsMap();
29}

31const QMap<QString, QString>& Config::commandLineArgs() const {
32 return _paramsMap;
33}

35QVariant Config::getValueImplementation(const QString& key, const QVariant& def)
36{
37 // 1. The command-line value has the highest priority.
38 if (_paramsMap.contains(key)) {
39 return _paramsMap.value(key);
40 }
41
42 // 2. The value that is saved in the config file (the Settings
43 // backend) and 3. the default value are resolved by the Settings
44 // base class. The def argument already contains the
45 // ISettings::defaultSettings value (see the ISettings::getValue
46 // method), so the default value has the lowest priority.
47 return Settings::getValueImplementation(key, def);
48}
