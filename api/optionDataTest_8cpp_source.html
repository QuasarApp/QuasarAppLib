45void OptionDataTest::testDeprecatedAndRemoved() {
46 // a plain option is not deprecated and not removed.
47 OptionData plainOption({"-plainOption"});
48 QVERIFY(!plainOption.isDeprecated());
49 QVERIFY(!plainOption.isRemoved());
50
51 // an option with a deprecation message is deprecated.
52 OptionData deprecatedOption({"-depOption"},
53 "(value)",
54 "Deprecated option",
55 "",
56 "This option is deprecated.");
57 QVERIFY(deprecatedOption.isDeprecated());
58 QCOMPARE(deprecatedOption.deprecatedMsg(),
59 QString("This option is deprecated."));
60 QVERIFY(!deprecatedOption.isRemoved());
61
62 // an option with the removed flag is removed.
63 OptionData removedOption({"-rmOption"},
64 "(value)",
65 "Removed option",
66 "",
67 "This option was removed.",
68 true);
69 QVERIFY(removedOption.isRemoved());
70 QVERIFY(removedOption.isDeprecated());
71}
